> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `skillsplit/stage1/README.md` 설계대로, SKILL.md를 정규식 규칙 16개로 검사해 팀 공통 형식(`{"skill_id","score","evidence"}`)으로 점수를 내는 정적 탐지기 v0와, dev에서 규칙 채택을 판단하는 보고서를 만든다.

**Architecture:** `rules.py`는 규칙 목록(정규식 + 심각도 + 범위 + 결합 조건)만 들고, `scan.py`는 SKILL.md를 frontmatter/prose/code 영역으로 나눈 "같은 길이의 가려진 문자열"에 규칙을 적용해 noisy-OR 점수를 낸다. 읽기·쓰기는 `stage2/common.py`를 import해 2단계와 같은 입력·출력을 쓴다. `data/s1_rule_report.py`가 dev 전체에서 규칙별 정상 적중률·출처 수를 내 채택을 판단한다.

**Tech Stack:** Python 3.12 표준 라이브러리(`re`, `bisect`, `argparse`, `dataclasses`), pandas(보고서만), pytest(테스트만).

## Global Constraints

- `skillsplit/stage2/` 파일은 수정하지 않는다. `skillsplit.stage2.common`의 `list_skills`, `read_skill`, `write_result`, `append_jsonl`만 import한다.
- 출력: 한 줄에 `{"skill_id": benchmark_id, "score": 0~1, "evidence": ["RULE@L줄번호", ...]}`, 부가 정보는 `<out>.details.jsonl` (`Path(out).with_suffix(".details.jsonl")`, 2단계와 같은 이름 규칙).
- 가중치: CRITICAL 0.9, HIGH 0.7, MEDIUM 0.4, LOW 0.2. `score = 1 − Π(1 − w)`, 걸린 규칙마다 한 번. 규칙마다 evidence 최대 3줄.
- oversize: 200,000자 초과면 `oversize=true`. 검사는 앞 300,000자까지.
- 정규식: `.*` 룩어헤드·중첩 반복 금지, 사이 간격은 `{0,N}` 상한. 모든 규칙은 import 때 컴파일.
- test 분할은 쓰지 않는다. 규칙 선택은 dev 전체(8,348), 2단계 비교는 dev100.
- 악성 원문을 출력하지 않는다. 테스트 문장은 직접 쓴 무해한 문장과 `example.invalid` 주소만 쓴다.

```python
import re
from dataclasses import dataclass

WEIGHT = {"CRITICAL": 0.9, "HIGH": 0.7, "MEDIUM": 0.4, "LOW": 0.2}


@dataclass(frozen=True)
class Rule:
    id: str
    severity: str          # WEIGHT의 키
    scope: str             # frontmatter | prose | code | all
    pattern: re.Pattern
    requires: tuple = ()   # 결합 전용: 이 중 하나가 함께 걸려야 발동. ("*",)는 결합 전용이 아닌 규칙 아무거나
    source: str = "skillsplit"


def _r(id, severity, scope, pattern, requires=(), source="skillsplit"):
    return Rule(id, severity, scope, re.compile(pattern, re.I), tuple(requires), source)


RULES = []
```

```python
"""1단계 정적 탐지기 v0: SKILL.md를 영역별로 나눠 규칙을 적용하고 noisy-OR 점수를 낸다.

  python -m skillsplit.stage1.scan data/dev100 --out runs/stage1_v0_dev100.jsonl
"""
import re

FENCE_OPEN = re.compile(r"^[ \t]{0,3}(`{3,}|~{3,})")


def _line_kinds(lines):
    """줄마다 frontmatter / prose / code."""
    kinds = ["prose"] * len(lines)
    start = 0
    if lines and lines[0].strip() == "---":
        for j in range(1, len(lines)):
            if lines[j].strip() == "---":
                kinds[: j + 1] = ["frontmatter"] * (j + 1)
                start = j + 1
                break
    fence = None
    for k in range(start, len(lines)):
        if fence is None:
            m = FENCE_OPEN.match(lines[k])
            if m:
                fence = m.group(1)
                kinds[k] = "code"
        else:
            kinds[k] = "code"
            t = lines[k].strip()
            if t and set(t) == {fence[0]} and len(t) >= len(fence):  # 같은 문자, 같거나 긴 펜스로만 닫힘
                fence = None
    return kinds


def segment(text):
    """영역별로 같은 길이의 문자열. 대상이 아닌 줄은 공백으로 덮어 원문 위치(줄 번호)를 유지한다."""
    lines = text.split("\n")
    kinds = _line_kinds(lines)
    views = {"all": text}
    for kind in ("frontmatter", "prose", "code"):
        views[kind] = "\n".join(s if k == kind else " " * len(s) for s, k in zip(lines, kinds))
    return views
```

파일 끝에 추가:

```python
def _active(rule, base):
    """결합 조건 확인. base = 걸린 규칙 중 결합 전용이 아닌 것의 id."""
    if not rule.requires:
        return True
    if rule.requires == ("*",):
        return bool(base)
    return bool(base & set(rule.requires))


def scan_text(text, rules=RULES):
    """(score, evidence, details)."""
    details = {"oversize": len(text) > OVERSIZE, "scanned_chars": min(len(text), SCAN_CAP), "empty": not text.strip()}
    text = text[:SCAN_CAP]
    views = segment(text)
    newlines = [m.start() for m in re.finditer("\n", text)]
    hits = {}
    for r in rules:
        lines = []
        for m in r.pattern.finditer(views[r.scope]):
            ln = bisect.bisect_right(newlines, m.start()) + 1
            if ln not in lines:
                lines.append(ln)
                if len(lines) == MAX_LINES_PER_RULE:
                    break
        if lines:
            hits[r.id] = lines
    by_id = {r.id: r for r in rules}
    base = {i for i in hits if not by_id[i].requires}
    fired = [r for r in rules if r.id in hits and _active(r, base)]
    keep = 1.0
    for r in fired:
        keep *= 1 - WEIGHT[r.severity]
    fired.sort(key=lambda r: (-WEIGHT[r.severity], hits[r.id][0]))
    evidence = [f"{r.id}@L{ln}" for r in fired for ln in hits[r.id]]
    return 1 - keep, evidence, details
```

```python
# 필요: dev_skills/(make_dev.py), runs_dev/scores.csv(E1), dev100_labels.csv(sample.py)
# 사용: cd data && python s1_rule_report.py → runs_dev/s1_v0_rules.csv, runs_dev/s1_v0_samples.csv
import sys
from pathlib import Path

import pandas as pd

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
from skillsplit.stage1.rules import RULES
from skillsplit.stage1.scan import scan_text
from skillsplit.stage2.common import read_skill

MIN_SOURCE_HITS = 3   # 출처 하나로 셀 최소 악성 적중 수
MAX_BENIGN_RATE = 0.01
MAX_FPR = 0.02


def pr_auc(y, s):
    """sklearn average_precision_score와 같은 계산(같은 점수는 한 문턱으로 묶음)."""
    d = pd.DataFrame({"y": y, "s": s}).groupby("s")["y"].agg(["sum", "count"]).iloc[::-1]
    tp, n = d["sum"].cumsum(), d["count"].cumsum()
    prec, rec = tp / n, tp / d["sum"].sum()
    return float(((rec - rec.shift(fill_value=0)) * prec).sum())


def run(m, rules):
    rows = []
    for b in m.benchmark_id:
        score, ev, _ = scan_text(read_skill(Path("dev_skills") / b)[1], rules)
        rows.append({"benchmark_id": b, "score": score, "rules": " ".join(sorted({e.split("@")[0] for e in ev}))})
    return m.merge(pd.DataFrame(rows), on="benchmark_id")


def summary(d, name):
    mal, ben = d[d.label == 1], d[d.label == 0]
    print(f"{name}: n={len(d)} PR-AUC={pr_auc(d.label, d.score):.3f} "
          f"재현율={(mal.score > 0).mean():.3f} 오탐률={(ben.score > 0).mean():.3f} ({(ben.score > 0).sum()}/{len(ben)})")


def main():
    m = pd.read_csv("runs_dev/scores.csv")
    d = run(m, RULES)
    d.to_csv("runs_dev/s1_v0_samples.csv", index=False)
    hit = lambda rid: d.rules.str.split().apply(lambda xs: rid in xs)

    rows = []
    for r in RULES:
        h = hit(r.id)
        mh, bh = d[h & (d.label == 1)], d[h & (d.label == 0)]
        src = mh.source_id.value_counts()
        n_src = int((src >= MIN_SOURCE_HITS).sum())
        ben_rate = len(bh) / (d.label == 0).sum()
        status = ("결합 전용/제외 검토" if ben_rate > MAX_BENIGN_RATE and not r.requires
                  else "채택" if n_src >= 2 else "출처 편향")
        rows.append({"rule": r.id, "severity": r.severity, "mal_hits": len(mh), "ben_hits": len(bh),
                     "ben_rate": round(ben_rate, 4), "n_sources": n_src,
                     "top_source": f"{src.index[0]} {src.iloc[0] / len(mh):.0%}" if len(mh) else "",
                     "new_vs_baselines": int(((mh.cisco_any == 0) & (mh.skillgate == 0)).sum()), "status": status})
    R = pd.DataFrame(rows)
    R.to_csv("runs_dev/s1_v0_rules.csv", index=False)
    print("== 규칙별 (dev 전체)\n" + R.to_string(index=False))

    print("\n== 전체")
    summary(d, "dev 전체, 규칙 전부")
    biased = set(R[R.status == "출처 편향"].rule)
    if biased:
        summary(run(m, [r for r in RULES if r.id not in biased]), f"dev 전체, 출처 편향 제외 {sorted(biased)}")
    summary(d[d.benchmark_id.isin(pd.read_csv("dev100_labels.csv").benchmark_id)], "dev100")
    fpr = (d[d.label == 0].score > 0).mean()
    print(f"오탐률 기준 {MAX_FPR:.0%}: {'통과' if fpr <= MAX_FPR else '초과 → 정밀도 낮은 규칙부터 뺀다'}")

    print("\n== 출처별 악성 재현율")
    print(d[d.label == 1].groupby("source_id").score.agg(n="count", recall=lambda s: round((s > 0).mean(), 3)).to_string())

    print("\n== Cisco(기본)와 네 칸 (악성)")
    a, c = d.score > 0, d.cisco_any == 1
    mal = d.label == 1
    print(f"정적만 {(a & ~c & mal).sum()}, Cisco만 {(~a & c & mal).sum()}, 둘 다 {(a & c & mal).sum()}, 둘 다 놓침 {(~a & ~c & mal).sum()}")


if __name__ == "__main__":
    main()
```
